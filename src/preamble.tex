% preamble.tex -- shared, ATS-safe settings for every resume. Compile with XeLaTeX.
% Rules: single column, no tables/icons/images, plain-text section names,
% no ligatures (fi/ff stay separate letters), no hyphenation (keywords never split).
\usepackage[a4paper,top=0.45in,bottom=0.45in,left=0.65in,right=0.65in]{geometry}
\usepackage{fontspec}
\setmainfont{Latin Modern Roman}[Ligatures=NoCommon, Mapping=tex-text, RawFeature={-liga;-clig;-dlig}]
\usepackage{enumitem}
\usepackage[hidelinks]{hyperref}
\usepackage{titlesec}
\usepackage{xspace}
\input{personal.tex}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}
\hyphenpenalty=10000
\exhyphenpenalty=10000
\emergencystretch=3em
\setlist[itemize]{leftmargin=1.1em, itemsep=0pt, topsep=1pt, parsep=0pt, partopsep=0pt}
\titleformat{\section}{\normalsize\bfseries}{}{0pt}{}[\vspace{-3pt}\titlerule]
\titlespacing*{\section}{0pt}{6pt}{3pt}

  \end{center}\vspace{-8pt}}
